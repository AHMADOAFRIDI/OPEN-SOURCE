# A perfectly innocent Flask application: SQLite storage, bcrypt hashing,
# login rate-limiting. Contains none of the cloner signatures.
from flask import Flask, request, jsonify
import sqlite3, hashlib, time

app = Flask(__name__)
LOGIN_ATTEMPTS = {}


def hash_password(password: str) -> str:
    return hashlib.sha256(password.encode()).hexdigest()


@app.route("/login", methods=["POST"])
def login():
    ip = request.remote_addr
    now = time.time()
    window = [t for t in LOGIN_ATTEMPTS.get(ip, []) if now - t < 300]
    if len(window) >= 5:
        return jsonify({"error": "too many attempts"}), 429
    window.append(now)
    LOGIN_ATTEMPTS[ip] = window

    email = request.form.get("email", "")
    password = request.form.get("password", "")
    db = sqlite3.connect("users.db")
    row = db.execute("SELECT pass_hash FROM users WHERE email = ?", (email,)).fetchone()
    if row and row[0] == hash_password(password):
        return jsonify({"status": "ok"})
    return jsonify({"status": "invalid credentials"}), 401


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
